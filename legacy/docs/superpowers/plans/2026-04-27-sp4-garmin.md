# SP4 — Garmin Connect Integration Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** OAuth 1.0a authentication with Garmin Connect, activity pull (creating WorkoutEntry records with LoggedExercise data), daily metrics pull (improving DerivedMetrics quality), and workout plan push (updating Garmin workouts when the local plan changes).

**Architecture:** `GarminAuth` handles the OAuth 1.0a flow via `ASWebAuthenticationSession` and stores tokens in iOS Keychain. `GarminAPIClient` is a thin actor that injects OAuth headers into every request. `GarminActivitySync` fetches activities and exercise sets, mapping to WorkoutEntry. `GarminDailySync` fetches daily summaries for DerivedMetrics. `GarminPlanPush` pushes plan changes to the Garmin workout service. `ExerciseCatalogue` maps local exercise names to validated Garmin name/category pairs. `ExerciseSubstitution` suggests substitutes by muscle group.

**Tech Stack:** `AuthenticationServices` (`ASWebAuthenticationSession`), Security framework (Keychain), `CryptoKit` (HMAC-SHA1), `BackgroundTasks`, Apple Testing framework.

**Prerequisites:** SP1 (WorkoutEntry), SP2 (BackgroundSyncManager), SP3a (Signal types), SP3b (DerivedMetrics).

**Known constraint:** Garmin exercise names must match Garmin's internal catalogue exactly. Unknown names are silently cleared by the API. The `ExerciseCatalogue` is the safeguard.

```swift
import Foundation

struct SubstitutionSuggestion {
    let entry: GarminExerciseEntry
    let reason: String
}

enum ExerciseSubstitution {

    /// Returns substitute exercises for the same primary muscle group.
    /// Excludes the original exercise and any already excluded by the caller.
    static func suggestions(for localName: String, excluding: [String] = []) -> [SubstitutionSuggestion] {
        guard let original = ExerciseCatalogue.resolve(localName) else { return [] }
        let primaryGroup = original.muscleGroups.first ?? .core
        let excluded = Set((excluding + [localName]).map { $0.lowercased() })

        return ExerciseCatalogue.entries
            .filter { $0.muscleGroups.contains(primaryGroup) && !excluded.contains($0.localName.lowercased()) }
            .map { SubstitutionSuggestion(entry: $0, reason: "Same muscle group: \(primaryGroup.rawValue)") }
    }
}
```

```swift
// JournalInsight/Garmin/GarminAuth.swift
import Foundation
import CryptoKit
import AuthenticationServices
import Security

struct GarminTokens: Codable {
    let accessToken: String
    let accessSecret: String
}

// Garmin's OAuth endpoints (OAuth 1.0a)
private enum GarminOAuthURL {
    static let requestToken  = "https://connectapi.garmin.com/oauth-service/oauth/request_token"
    static let authorize     = "https://connect.garmin.com/oauthConfirm"
    static let accessToken   = "https://connectapi.garmin.com/oauth-service/oauth/access_token"
    static let callbackScheme = "journalinsight"
    static let callbackURL    = "journalinsight://garmin/callback"
}

// Replace with your registered Garmin Connect API consumer key/secret.
// Store these in a Config.plist excluded from source control, or use environment variables.
private enum GarminConsumer {
    static let key    = Bundle.main.infoDictionary?["GARMIN_CONSUMER_KEY"] as? String ?? ""
    static let secret = Bundle.main.infoDictionary?["GARMIN_CONSUMER_SECRET"] as? String ?? ""
}

enum GarminAuth {

    // MARK: — Public API

    /// Initiates OAuth flow via ASWebAuthenticationSession.
    /// Stores tokens in Keychain on success.
    @MainActor
    static func connect(presentationAnchor: ASPresentationAnchor) async throws {
        // Step 1: Get request token
        let (requestToken, requestSecret) = try await fetchRequestToken()

        // Step 2: User authorises in browser
        let callbackURL = try await authorize(requestToken: requestToken, anchor: presentationAnchor)

        // Step 3: Exchange for access token
        guard let verifier = URLComponents(url: callbackURL, resolvingAgainstBaseURL: false)?
            .queryItems?.first(where: { $0.name == "oauth_verifier" })?.value
        else { throw GarminAuthError.missingVerifier }

        let tokens = try await fetchAccessToken(
            requestToken: requestToken, requestSecret: requestSecret, verifier: verifier)
        try storeTokens(tokens)
    }

    static func loadTokens() -> GarminTokens? {
        guard let data = keychainRead(key: keychainKey),
              let tokens = try? JSONDecoder().decode(GarminTokens.self, from: data)
        else { return nil }
        return tokens
    }

    static func clearTokens() {
        keychainDelete(key: keychainKey)
    }

    // MARK: — OAuth helpers (testable)

    static func hmacSHA1(key: String, message: String) -> String {
        let keyData = SymmetricKey(data: Data(key.utf8))
        let mac = HMAC<Insecure.SHA1>.authenticationCode(for: Data(message.utf8), using: keyData)
        return Data(mac).base64EncodedString()
    }

    static func oauthEncode(_ value: String) -> String {
        value.addingPercentEncoding(
            withAllowedCharacters: CharacterSet(charactersIn:
                "ABCDEFGHIJKLMNOPQRSTUVWXYZabcdefghijklmnopqrstuvwxyz0123456789-._~")
        ) ?? value
    }

    static func generateNonce() -> String {
        let chars = "abcdefghijklmnopqrstuvwxyzABCDEFGHIJKLMNOPQRSTUVWXYZ0123456789"
        return String((0..<20).compactMap { _ in chars.randomElement() })
    }

    // Builds the Authorization header value for a signed request
    static func authorizationHeader(
        method: String, url: String,
        params: [String: String] = [:],
        token: String = "", tokenSecret: String = ""
    ) -> String {
        let timestamp = String(Int(Date.now.timeIntervalSince1970))
        let nonce = generateNonce()

        var oauthParams: [String: String] = [
            "oauth_consumer_key":     GarminConsumer.key,
            "oauth_nonce":            nonce,
            "oauth_signature_method": "HMAC-SHA1",
            "oauth_timestamp":        timestamp,
            "oauth_version":          "1.0",
        ]
        if !token.isEmpty { oauthParams["oauth_token"] = token }

        let allParams = oauthParams.merging(params) { o, _ in o }
        let paramString = allParams.sorted(by: { $0.key < $1.key })
            .map { "\(oauthEncode($0.key))=\(oauthEncode($0.value))" }
            .joined(separator: "&")

        let base = "\(method)&\(oauthEncode(url))&\(oauthEncode(paramString))"
        let sigKey = "\(oauthEncode(GarminConsumer.secret))&\(oauthEncode(tokenSecret))"
        let signature = hmacSHA1(key: sigKey, message: base)

        oauthParams["oauth_signature"] = signature
        let header = oauthParams.sorted(by: { $0.key < $1.key })
            .map { "\($0.key)=\"\(oauthEncode($0.value))\"" }
            .joined(separator: ", ")
        return "OAuth \(header)"
    }

    // MARK: — Private

    private static func fetchRequestToken() async throws -> (token: String, secret: String) {
        let auth = authorizationHeader(method: "POST", url: GarminOAuthURL.requestToken,
                                       params: ["oauth_callback": GarminOAuthURL.callbackURL])
        var req = URLRequest(url: URL(string: GarminOAuthURL.requestToken)!)
        req.httpMethod = "POST"
        req.setValue(auth, forHTTPHeaderField: "Authorization")
        let (data, _) = try await URLSession.shared.data(for: req)
        let body = String(data: data, encoding: .utf8) ?? ""
        let pairs = parseQueryString(body)
        guard let token = pairs["oauth_token"], let secret = pairs["oauth_token_secret"]
        else { throw GarminAuthError.badRequestToken }
        return (token, secret)
    }

    @MainActor
    private static func authorize(requestToken: String, anchor: ASPresentationAnchor) async throws -> URL {
        let authURL = URL(string: "\(GarminOAuthURL.authorize)?oauth_token=\(requestToken)")!
        return try await withCheckedThrowingContinuation { continuation in
            let session = ASWebAuthenticationSession(
                url: authURL, callbackURLScheme: GarminOAuthURL.callbackScheme
            ) { url, error in
                if let url { continuation.resume(returning: url) }
                else { continuation.resume(throwing: error ?? GarminAuthError.authCancelled) }
            }
            session.presentationContextProvider = PresentationProvider(anchor: anchor)
            session.prefersEphemeralWebBrowserSession = false
            session.start()
        }
    }

    private static func fetchAccessToken(requestToken: String, requestSecret: String, verifier: String) async throws -> GarminTokens {
        let params = ["oauth_verifier": verifier]
        let auth = authorizationHeader(method: "POST", url: GarminOAuthURL.accessToken,
                                       params: params, token: requestToken, tokenSecret: requestSecret)
        var req = URLRequest(url: URL(string: GarminOAuthURL.accessToken)!)
        req.httpMethod = "POST"
        req.setValue(auth, forHTTPHeaderField: "Authorization")
        req.httpBody = "oauth_verifier=\(verifier)".data(using: .utf8)
        let (data, _) = try await URLSession.shared.data(for: req)
        let body = String(data: data, encoding: .utf8) ?? ""
        let pairs = parseQueryString(body)
        guard let token = pairs["oauth_token"], let secret = pairs["oauth_token_secret"]
        else { throw GarminAuthError.badAccessToken }
        return GarminTokens(accessToken: token, accessSecret: secret)
    }

    private static func parseQueryString(_ s: String) -> [String: String] {
        var result: [String: String] = [:]
        for pair in s.components(separatedBy: "&") {
            let parts = pair.components(separatedBy: "=")
            if parts.count == 2 { result[parts[0]] = parts[1].removingPercentEncoding }
        }
        return result
    }

    private static func storeTokens(_ tokens: GarminTokens) throws {
        guard let data = try? JSONEncoder().encode(tokens) else { throw GarminAuthError.keychainFailed }
        keychainWrite(key: keychainKey, data: data)
    }

    // MARK: — Keychain helpers

    private static let keychainKey = "garminTokens"

    private static func keychainWrite(key: String, data: Data) {
        let query: [CFString: Any] = [
            kSecClass: kSecClassGenericPassword,
            kSecAttrAccount: key,
            kSecValueData: data,
        ]
        SecItemDelete(query as CFDictionary)
        SecItemAdd(query as CFDictionary, nil)
    }

    private static func keychainRead(key: String) -> Data? {
        let query: [CFString: Any] = [
            kSecClass: kSecClassGenericPassword,
            kSecAttrAccount: key,
            kSecReturnData: true,
            kSecMatchLimit: kSecMatchLimitOne,
        ]
        var result: AnyObject?
        SecItemCopyMatching(query as CFDictionary, &result)
        return result as? Data
    }

    private static func keychainDelete(key: String) {
        let query: [CFString: Any] = [kSecClass: kSecClassGenericPassword, kSecAttrAccount: key]
        SecItemDelete(query as CFDictionary)
    }
}

// Provides ASPresentationAnchor for ASWebAuthenticationSession
private class PresentationProvider: NSObject, ASWebAuthenticationPresentationContextProviding {
    let anchor: ASPresentationAnchor
    init(anchor: ASPresentationAnchor) { self.anchor = anchor }
    func presentationAnchor(for session: ASWebAuthenticationSession) -> ASPresentationAnchor { anchor }
}

enum GarminAuthError: Error {
    case badRequestToken, missingVerifier, badAccessToken, authCancelled, keychainFailed
}
```

```swift
// JournalInsight/Garmin/GarminAPIClient.swift
import Foundation

// All Garmin Connect API calls go through connectapi.garmin.com — NOT connect.garmin.com.
// Calling connect.garmin.com with PUT returns 405. connectapi.garmin.com is the correct host.
private let baseURL = "https://connectapi.garmin.com"

actor GarminAPIClient {
    private let tokens: GarminTokens

    init(tokens: GarminTokens) {
        self.tokens = tokens
    }

    // MARK: — GET

    func get(_ path: String) async throws -> Data {
        let url = URL(string: baseURL + path)!
        var req = URLRequest(url: url)
        req.httpMethod = "GET"
        req.setValue(authHeader(method: "GET", url: url.absoluteString), forHTTPHeaderField: "Authorization")
        req.setValue("application/json", forHTTPHeaderField: "Accept")
        let (data, resp) = try await URLSession.shared.data(for: req)
        try checkResponse(resp, data: data)
        return data
    }

    // MARK: — PUT

    func put(_ path: String, body: [String: Any]) async throws {
        let url = URL(string: baseURL + path)!
        let bodyData = try JSONSerialization.data(withJSONObject: body)
        var req = URLRequest(url: url)
        req.httpMethod = "PUT"
        req.setValue(authHeader(method: "PUT", url: url.absoluteString), forHTTPHeaderField: "Authorization")
        req.setValue("application/json", forHTTPHeaderField: "Content-Type")
        req.httpBody = bodyData
        let (_, resp) = try await URLSession.shared.data(for: req)
        try checkResponse(resp, data: nil)
    }

    // MARK: — Private

    private func authHeader(method: String, url: String) -> String {
        GarminAuth.authorizationHeader(
            method: method, url: url,
            token: tokens.accessToken, tokenSecret: tokens.accessSecret
        )
    }

    private func checkResponse(_ response: URLResponse, data: Data?) throws {
        guard let http = response as? HTTPURLResponse else { return }
        switch http.statusCode {
        case 200...204: return
        case 401, 403: throw GarminAPIError.authExpired
        case 404:       throw GarminAPIError.notFound
        default:
            let body = data.flatMap { String(data: $0, encoding: .utf8) } ?? ""
            throw GarminAPIError.httpError(http.statusCode, body)
        }
    }
}

enum GarminAPIError: Error {
    case authExpired
    case notFound
    case httpError(Int, String)
}
```

```swift
// JournalInsight/Garmin/GarminActivitySync.swift
import Foundation
import SwiftData

actor GarminActivitySync {
    private let client: GarminAPIClient
    private let modelContext: ModelContext

    init(client: GarminAPIClient, modelContext: ModelContext) {
        self.client = client
        self.modelContext = modelContext
    }

    /// Pull activities from last `days` days and create WorkoutEntry records.
    func sync(days: Int = 30) async throws {
        let since = Calendar.current.date(byAdding: .day, value: -days, to: .now)!
        let until = Date.now
        let sinceStr = ISO8601DateFormatter().string(from: since).prefix(10)
        let untilStr = ISO8601DateFormatter().string(from: until).prefix(10)

        let path = "/activitylist-service/activities/search/activities?startDate=\(sinceStr)&endDate=\(untilStr)&limit=999&start=0"
        let data = try await client.get(path)
        let activities = try parseActivities(data)

        for act in activities {
            guard let actId = act["activityId"] as? Int64 else { continue }
            let sport = (act["activityType"] as? [String: Any])?["typeKey"] as? String ?? ""
            guard sport == "strength_training" || sport == "multi_sport" else { continue }

            let effectiveId: Int64
            if sport == "multi_sport" {
                effectiveId = (try? await resolveStrengthChild(parentId: actId)) ?? actId
            } else {
                effectiveId = actId
            }

            let sets = (try? await fetchExerciseSets(activityId: effectiveId)) ?? []
            await MainActor.run { insertOrUpdate(activity: act, activityId: effectiveId, sets: sets) }
        }
    }

    // MARK: — Private

    private func parseActivities(_ data: Data) throws -> [[String: Any]] {
        let json = try JSONSerialization.jsonObject(with: data)
        if let arr = json as? [[String: Any]] { return arr }
        if let dict = json as? [String: Any] {
            return dict["activityList"] as? [[String: Any]] ?? []
        }
        return []
    }

    private func resolveStrengthChild(parentId: Int64) async throws -> Int64? {
        let data = try await client.get("/activity-service/activity/\(parentId)")
        guard let json = try? JSONSerialization.jsonObject(with: data) as? [String: Any],
              let meta = json["metadataDTO"] as? [String: Any],
              let childIds = meta["childIds"] as? [Int64],
              let childTypes = meta["childActivityTypes"] as? [String]
        else { return nil }
        return zip(childIds, childTypes).first { $0.1 == "strength_training" }?.0
    }

    private func fetchExerciseSets(activityId: Int64) async throws -> [LoggedExercise] {
        let data = try await client.get("/activity-service/activity/\(activityId)/exerciseSets")
        guard let json = try? JSONSerialization.jsonObject(with: data) as? [String: Any],
              let sets = json["exerciseSets"] as? [[String: Any]]
        else { return [] }

        return sets.compactMap { s -> LoggedExercise? in
            guard (s["setType"] as? String) == "ACTIVE" else { return nil }
            let exArr = s["exercises"] as? [[String: Any]] ?? []
            let ex = exArr.first ?? [:]
            let cat = ex["category"] as? String ?? (s["category"] as? String ?? "")
            let name = ex["name"] as? String ?? (s["exerciseName"] as? String)
            let reps = s["repetitionCount"] as? Int ?? (s["repetitions"] as? Int ?? 0)
            let weightG = s["weight"] as? Double
            let weightKg = weightG.map { $0 / 1000 }

            // Reverse-lookup local name from Garmin catalogue
            let localName = ExerciseCatalogue.entries
                .first { $0.garminCategory == cat && $0.garminName == name }?.localName
                ?? name ?? cat

            return LoggedExercise(name: localName, sets: 1, reps: reps,
                                  weightKg: weightKg, garminExerciseName: name)
        }
    }

    @MainActor
    private func insertOrUpdate(activity: [String: Any], activityId: Int64, sets: [LoggedExercise]) {
        let descriptor = FetchDescriptor<WorkoutEntry>(
            predicate: #Predicate { $0.garminActivityId == activityId }
        )
        if (try? modelContext.fetchCount(descriptor)) ?? 0 > 0 { return }

        let startStr = (activity["startTimeLocal"] as? String) ?? ""
        let date = ISO8601DateFormatter().date(from: startStr) ?? .now
        let duration = Int(activity["duration"] as? Double ?? 0)

        let entry = WorkoutEntry(
            date: date, source: .garmin, durationSec: duration,
            exercises: sets, garminActivityId: activityId
        )
        modelContext.insert(entry)
        try? modelContext.save()
    }
}
```

```swift
// JournalInsight/Garmin/GarminDailySync.swift
import Foundation

/// Fetches Garmin daily summaries and sleep data for DerivedMetrics.
/// Returns rows in the same format as HealthTraining's load_daily_metrics().
actor GarminDailySync {
    private let client: GarminAPIClient

    init(client: GarminAPIClient) { self.client = client }

    func fetchDailyRows(days: Int = 7) async throws -> [[String: Double?]] {
        let cal = Calendar.current
        let since = cal.date(byAdding: .day, value: -days, to: .now)!
        let until = Date.now

        async let dailyData = fetchDailySummaries(since: since, until: until)
        async let sleepData = fetchSleepSummaries(since: since, until: until)

        let daily = try await dailyData
        let sleep = try await sleepData

        // Merge by date
        var byDate: [String: [String: Double?]] = [:]
        for d in daily {
            if let date = d["date"] as? String { byDate[date] = buildRow(from: d) }
        }
        for s in sleep {
            if let date = s["date"] as? String {
                byDate[date, default: [:]]["sleep_score"] = s["sleep_score"] as? Double
            }
        }

        return byDate.values.sorted { ($0["date"] as? String ?? "") < ($1["date"] as? String ?? "") }
            .map { $0 as! [String: Double?] }
    }

    // MARK: — Private

    private func fetchDailySummaries(since: Date, until: Date) async throws -> [[String: Any]] {
        let fmt = DateFormatter(); fmt.dateFormat = "yyyy-MM-dd"
        let start = fmt.string(from: since), end = fmt.string(from: until)
        let path = "/usersummary-service/usersummary/dailysummaries?startDate=\(start)&endDate=\(end)"
        let data = try await client.get(path)
        guard let json = try? JSONSerialization.jsonObject(with: data) as? [String: Any],
              let arr = json["userDailySummaries"] as? [[String: Any]]
        else { return [] }
        return arr
    }

    private func fetchSleepSummaries(since: Date, until: Date) async throws -> [[String: Any]] {
        let fmt = DateFormatter(); fmt.dateFormat = "yyyy-MM-dd"
        let start = fmt.string(from: since), end = fmt.string(from: until)
        let path = "/sleep-service/sleep/dailySleepSummariesByDate?startDate=\(start)&endDate=\(end)"
        let data = try await client.get(path)
        guard let json = try? JSONSerialization.jsonObject(with: data) as? [String: Any],
              let arr = json["dailySleepSummaries"] as? [[String: Any]]
        else { return [] }
        return arr
    }

    private func buildRow(from d: [String: Any]) -> [String: Double?] {
        [
            "date":               d["calendarDate"] as? Double,
            "rhr_bpm":            (d["restingHeartRate"] as? Double),
            "steps":              (d["totalSteps"] as? Double),
            "kcal_burned_active": (d["activeKilocalories"] as? Double),
            "body_battery_avg":   (d["bodyBatteryMostRecentValue"] as? Double),
        ]
    }
}
```

```swift
// JournalInsight/Garmin/GarminPlanPush.swift
import Foundation

actor GarminPlanPush {
    private let client: GarminAPIClient

    init(client: GarminAPIClient) { self.client = client }

    /// Push an updated exercise list to an existing Garmin workout.
    /// workoutId: the Garmin workout ID (from the Garmin Connect workout).
    /// exercises: list of exercises with validated Garmin names.
    func pushPlanUpdate(workoutId: Int64, workoutName: String, exercises: [PlanExercise]) async throws {
        let steps = exercises.flatMap { buildSteps(from: $0) }
        let payload: [String: Any] = [
            "workoutId":   workoutId,
            "workoutName": workoutName,
            "sportType":   ["sportTypeKey": "strength_training"],
            "workoutSegments": [[
                "segmentOrder": 1,
                "sportType": ["sportTypeKey": "strength_training"],
                "workoutSteps": steps,
            ]],
        ]
        // PUT to connectapi.garmin.com — returns 204 (nil body) on success
        try await client.put("/workout-service/workout/\(workoutId)", body: payload)
    }

    // MARK: — Private

    private func buildSteps(from exercise: PlanExercise) -> [[String: Any]] {
        let catalogue = ExerciseCatalogue.resolve(exercise.name)
        var steps: [[String: Any]] = []

        // One ACTIVE step per set
        for setNum in 1...max(1, exercise.sets) {
            var step: [String: Any] = [
                "stepOrder":  steps.count + 1,
                "stepType":   ["stepTypeKey": "ACTIVE"],
                "targetType": ["workoutTargetTypeKey": "NO_TARGET"],
                "numberOfIterations": exercise.reps,
                "category":   catalogue?.garminCategory ?? "OTHER",
            ]
            // Only include exerciseName when it's a validated catalogue name
            if let garminName = catalogue?.garminName {
                step["exerciseName"] = garminName
            }
            if let kg = exercise.weightKg, kg > 0 {
                step["weight"] = Int(kg * 1000)  // Garmin stores weight in grams
                step["weightUnit"] = ["weightUnitKey": "KILOGRAM"]
            }
            step["description"] = "Set \(setNum)"
            steps.append(step)

            // REST step after each set (except the last)
            if setNum < exercise.sets {
                steps.append([
                    "stepOrder": steps.count + 1,
                    "stepType":  ["stepTypeKey": "REST"],
                    "targetType": ["workoutTargetTypeKey": "NO_TARGET"],
                    "durationValue": 90,
                    "durationType": ["durationTypeKey": "TIME"],
                ])
            }
        }
        return steps
    }
}

struct PlanExercise {
    let name: String        // local exercise name (resolved via ExerciseCatalogue)
    let sets: Int
    let reps: Int
    let weightKg: Double?
}
```
